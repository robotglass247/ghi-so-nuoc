function waterCanReplaceV875_(r,assessed,old,target,replaceId){
  if(!(assessed.safe||assessed.replaceSafe)||!replaceId||replaceId===String(r[18]))return false;

  const matches=a=>
    a &&
    String(a[1])===String(r[1]) &&
    chuanHoaMaDongHo_(a[5])===chuanHoaMaDongHo_(r[5]);

  // replaceId vẫn phải trỏ tới một ảnh có thật của đúng kỳ + đúng đồng hồ.
  if(!matches(target)||String(target[18])!==replaceId)return false;

  // Chưa có bản chính: ảnh thay thế hợp lệ được phép lên bản chính.
  if(!old||!old[17])return true;

  // Bản chính hiện tại phải cùng kỳ + cùng mã đồng hồ.
  if(!matches(old))return false;

  // Retry sau khi lần ghi trước mới hoàn thành một phần.
  if(old[17]===r[17])return true;

  // Chống ảnh cũ/offline ghi đè bản mới.
  const newer=new Date(r[0]).getTime();
  const before=new Date(old[0]).getTime();
  if(!Number.isFinite(newer)||!Number.isFinite(before)||newer<before)return false;

  // LATEST CAPTURE WINS:
  // Không bắt buộc replaceId phải chính là Client ID của bản chính hiện tại.
  // Trường hợp người dùng chụp lại nhiều lần liên tiếp:
  //   bản chính 998 -> ảnh lỗi 851 -> ảnh lỗi 110 -> ảnh đúng 1193
  // thì 1193 vẫn được promote nếu nó là ảnh mới hơn và AI đọc hợp lệ.
  return true;
}
