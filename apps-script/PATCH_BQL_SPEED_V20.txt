THAY TOÀN BỘ HÀM TRÊN BẰNG:

function waterBqlLastRowV20_(sh){
  const max = sh.getMaxRows();
  if (max < 2) return 1;

  const bottom = sh.getRange(max, 6); // cột F = Mã đồng hồ
  const bottomValue = waterBqlTextV20_(bottom.getDisplayValue());
  if (bottomValue) return max;

  const row = bottom
    .getNextDataCell(SpreadsheetApp.Direction.UP)
    .getRow();

  if (row < 2) return 1;

  return waterBqlTextV20_(sh.getRange(row, 6).getDisplayValue())
    ? row
    : 1;
}
