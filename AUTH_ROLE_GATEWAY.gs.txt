/* ============================================================
AUTH ROLE GATEWAY - STANDALONE APPS SCRIPT PROJECT
VERSION: 1.1.1-DEV
PURPOSE:
  1) Verify Google Identity Services ID token server-side.
  2) Match verified email with NHAN_SU_THUC_HIEN.
  3) Return role: QUẢN LÝ / NHÂN VIÊN.
  4) DEV transport via hidden form POST + iframe postMessage.
IMPORTANT:
  - Deploy as a SEPARATE Apps Script Web App.
  - DO NOT paste into current PASS Ma.gs project.
  - AUTH_ROLE_SETUP only prepares column G and removes old ACL trigger.
  - It does NOT change Google Drive sharing during Phase 1 testing.
============================================================ */

const AUTH_SHEET_ID = '1YeXaSA03l3wPntaP_aNKeR_aMrjCnenHtLAiALSwxpY';
const AUTH_STAFF_SHEET = 'NHAN_SU_THUC_HIEN';
const AUTH_GOOGLE_CLIENT_ID = '617076693631-mc3c0kf70ngdd96jjnn6qk3g39jk2v6o.apps.googleusercontent.com';
const AUTH_ALLOWED_PARENT_ORIGIN = 'https://robotglass247.github.io';
const AUTH_ADDED_EMAILS_PROP = 'AUTH_ROLE_ADDED_FILE_USERS_V1';

function doGet() {
  return authJson_({
    ok: true,
    service: 'AUTH_ROLE_GATEWAY',
    version: '1.1.1-DEV'
  });
}

function doPost(e) {
  const nonce = String(e && e.parameter && e.parameter.nonce || '').trim();
  let result;

  try {
    const credential = String(e && e.parameter && e.parameter.credential || '').trim();

    if (!credential) {
      throw new Error('Thiếu Google ID token.');
    }
    if (!AUTH_GOOGLE_CLIENT_ID) {
      throw new Error('AUTH Gateway chưa cấu hình Google Client ID.');
    }

    const identity = authVerifyGoogleIdToken_(credential);
    const access = authLookupStaff_(identity.email);

    result = {
      ok: true,
      authenticated: true,
      authorized: access.authorized,
      email: identity.email,
      name: access.name || identity.name || '',
      role: access.role || '',
      status: access.status || '',
      reason: access.reason || ''
    };
  } catch (err) {
    result = {
      ok: false,
      error: String(err && err.message || err)
    };
  }

  return authIframeResult_(result, nonce);
}

function authIframeResult_(payload, nonce) {
  const safePayload = JSON.stringify(payload)
    .replace(/</g, '\\u003c')
    .replace(/>/g, '\\u003e')
    .replace(/&/g, '\\u0026');
  const safeNonce = JSON.stringify(String(nonce || ''));
  const safeOrigin = JSON.stringify(AUTH_ALLOWED_PARENT_ORIGIN);

  const html = '<!doctype html><html><head><meta charset="utf-8"></head><body>' +
    '<script>(function(){try{' +
    'window.parent.postMessage({' +
      'type:"WATER_AUTH_GATEWAY_RESULT",' +
      'nonce:' + safeNonce + ',' +
      'payload:' + safePayload +
    '},' + safeOrigin + ');' +
    '}catch(e){}})();</' + 'script>' +
    '</body></html>';

  return HtmlService
    .createHtmlOutput(html)
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function authVerifyGoogleIdToken_(credential) {
  const url = 'https://oauth2.googleapis.com/tokeninfo?id_token=' + encodeURIComponent(credential);
  const response = UrlFetchApp.fetch(url, {
    method: 'get',
    muteHttpExceptions: true
  });

  if (response.getResponseCode() !== 200) {
    throw new Error('Google ID token không hợp lệ hoặc đã hết hạn.');
  }

  const p = JSON.parse(response.getContentText() || '{}');
  const now = Math.floor(Date.now() / 1000);
  const aud = String(p.aud || '');
  const iss = String(p.iss || '');
  const email = String(p.email || '').trim().toLowerCase();
  const verified = String(p.email_verified || '').toLowerCase() === 'true';
  const exp = Number(p.exp || 0);

  if (aud !== AUTH_GOOGLE_CLIENT_ID) {
    throw new Error('Google Client ID không khớp.');
  }
  if (iss !== 'accounts.google.com' && iss !== 'https://accounts.google.com') {
    throw new Error('Google token issuer không hợp lệ.');
  }
  if (!verified) {
    throw new Error('Email Google chưa được xác minh.');
  }
  if (!email) {
    throw new Error('Không lấy được email Google.');
  }
  if (!exp || exp <= now) {
    throw new Error('Google ID token đã hết hạn.');
  }

  return {
    email: email,
    name: String(p.name || '').trim(),
    sub: String(p.sub || '').trim()
  };
}

function authLookupStaff_(email) {
  const ss = SpreadsheetApp.openById(AUTH_SHEET_ID);
  const sh = ss.getSheetByName(AUTH_STAFF_SHEET);
  if (!sh) throw new Error('Không tìm thấy sheet ' + AUTH_STAFF_SHEET + '.');

  const lastRow = Math.max(sh.getLastRow(), 5);
  const values = sh.getRange(5, 1, lastRow - 4, 7).getDisplayValues();
  const wanted = String(email || '').trim().toLowerCase();

  for (let i = 0; i < values.length; i++) {
    const row = values[i];
    const code = String(row[0] || '').trim();
    const name = String(row[1] || '').trim();
    const staffEmail = String(row[4] || '').trim().toLowerCase();
    const status = String(row[5] || '').trim();
    const role = String(row[6] || '').trim();

    if (!code || !staffEmail || staffEmail !== wanted) continue;

    const active = authNorm_(status) === 'dang lam viec';
    const r = authNorm_(role);
    const roleValid = r === 'quan ly' || r === 'nhan vien';

    if (!active) {
      return {
        authorized:false,
        name:name,
        role:role,
        status:status,
        reason:'Tài khoản nhân sự không ở trạng thái Đang làm việc.'
      };
    }

    if (!roleValid) {
      return {
        authorized:false,
        name:name,
        role:role,
        status:status,
        reason:'Tài khoản chưa được phân quyền QUẢN LÝ/NHÂN VIÊN.'
      };
    }

    return {
      authorized:true,
      name:name,
      role:role,
      status:status,
      reason:''
    };
  }

  return {
    authorized:false,
    name:'',
    role:'',
    status:'',
    reason:'Email Google không có trong NHAN_SU_THUC_HIEN.'
  };
}

function authNorm_(v) {
  return String(v == null ? '' : v)
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/đ/g, 'd')
    .replace(/Đ/g, 'D')
    .replace(/\s+/g, ' ')
    .trim()
    .toLowerCase();
}

function authJson_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/* ============================================================
PHASE 1 - ROLE SETUP ONLY
Removes any legacy Drive-sync trigger from older DEV builds.
Does not modify Google Drive sharing.
============================================================ */
function AUTH_ROLE_SETUP() {
  const ss = SpreadsheetApp.openById(AUTH_SHEET_ID);
  const sh = ss.getSheetByName(AUTH_STAFF_SHEET);
  if (!sh) throw new Error('Không tìm thấy sheet ' + AUTH_STAFF_SHEET + '.');

  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (t.getHandlerFunction() === 'AUTH_ROLE_ON_EDIT') {
      ScriptApp.deleteTrigger(t);
    }
  });

  sh.getRange('G4').setValue('PHÂN QUYỀN');

  const maxRows = Math.max(sh.getMaxRows(), 5);
  const rule = SpreadsheetApp.newDataValidation()
    .requireValueInList(['QUẢN LÝ', 'NHÂN VIÊN'], true)
    .setAllowInvalid(false)
    .build();
  sh.getRange(5, 7, maxRows - 4, 1).setDataValidation(rule);

  return {ok:true, message:'Đã tạo cột PHÂN QUYỀN và gỡ trigger Drive cũ. Chưa thay đổi quyền Google Drive.'};
}

/* ============================================================
PHASE 2 - OPTIONAL DRIVE ACL SYNC
Run only AFTER role-auth DEV passes.
Only viewers added by this gateway are later removed by this gateway.
============================================================ */
function AUTH_ROLE_ENABLE_DRIVE_SYNC() {
  const exists = ScriptApp.getProjectTriggers().some(function(t) {
    return t.getHandlerFunction() === 'AUTH_ROLE_ON_EDIT';
  });

  if (!exists) {
    ScriptApp.newTrigger('AUTH_ROLE_ON_EDIT')
      .forSpreadsheet(AUTH_SHEET_ID)
      .onEdit()
      .create();
  }

  return AUTH_ROLE_SYNC_DRIVE();
}

function AUTH_ROLE_ON_EDIT(e) {
  try {
    const r = e && e.range;
    if (!r) return;
    const sh = r.getSheet();
    if (sh.getName() !== AUTH_STAFF_SHEET) return;
    if (r.getLastRow() < 5) return;
    if (r.getColumn() > 7 || r.getLastColumn() < 5) return;
    AUTH_ROLE_SYNC_DRIVE();
  } catch (err) {
    console.error(err);
  }
}

function AUTH_ROLE_SYNC_DRIVE() {
  const ss = SpreadsheetApp.openById(AUTH_SHEET_ID);
  const sh = ss.getSheetByName(AUTH_STAFF_SHEET);
  if (!sh) throw new Error('Không tìm thấy sheet ' + AUTH_STAFF_SHEET + '.');

  const lastRow = Math.max(sh.getLastRow(), 5);
  const rows = sh.getRange(5, 1, lastRow - 4, 7).getDisplayValues();
  const desired = [];

  rows.forEach(function(row) {
    const code = String(row[0] || '').trim();
    const email = String(row[4] || '').trim().toLowerCase();
    const active = authNorm_(row[5]) === 'dang lam viec';
    const manager = authNorm_(row[6]) === 'quan ly';
    if (code && email && active && manager && desired.indexOf(email) < 0) {
      desired.push(email);
    }
  });

  const file = DriveApp.getFileById(AUTH_SHEET_ID);
  const props = PropertiesService.getScriptProperties();

  let previouslyAdded = [];
  try {
    previouslyAdded = JSON.parse(props.getProperty(AUTH_ADDED_EMAILS_PROP) || '[]');
  } catch (e) {
    previouslyAdded = [];
  }

  const currentViewers = file.getViewers().map(function(u){return String(u.getEmail() || '').toLowerCase();});
  const currentEditors = file.getEditors().map(function(u){return String(u.getEmail() || '').toLowerCase();});
  const owner = file.getOwner();
  const ownerEmail = owner ? String(owner.getEmail() || '').toLowerCase() : '';

  const addedNow = [];

  desired.forEach(function(email) {
    const alreadyHasAccess = email === ownerEmail || currentViewers.indexOf(email) >= 0 || currentEditors.indexOf(email) >= 0;
    if (alreadyHasAccess) return;
    try {
      file.addViewer(email);
      addedNow.push(email);
    } catch (e) {
      console.error('addViewer', email, e);
    }
  });

  previouslyAdded.forEach(function(email) {
    if (desired.indexOf(email) >= 0) return;
    try { file.removeViewer(email); } catch (e) {}
  });

  const nextManaged = previouslyAdded
    .filter(function(email){return desired.indexOf(email) >= 0;})
    .concat(addedNow)
    .filter(function(email,index,arr){return arr.indexOf(email) === index;});

  props.setProperty(AUTH_ADDED_EMAILS_PROP, JSON.stringify(nextManaged));

  return {
    managers: desired.length,
    addedByGateway: nextManaged
  };
}
